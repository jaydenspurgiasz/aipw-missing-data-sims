\[
\begin{aligned}
\widehat{\mathbb E}[X_j]
&=\frac1n\sum_{i=1}^n\left[\widehat m_{ij}
+\frac{R_{ij}}{\widehat\pi_i}(X_{ij}-\widehat m_{ij})\right],\\
\widehat{\mathbb E}[X_jX_k]
&=\frac1n\sum_{i=1}^n\left[\widehat q_{ijk}
